cmake_minimum_required(VERSION 3.22)
project(bricohams_modules LANGUAGES C)

option(BRICOHAMS_ENABLE_ARDF "Build the independent ARDF timing and gain core" ON)
option(BRICOHAMS_ENABLE_SPECTRUM "Build the independent spectrum acquisition core" ON)
option(BRICOHAMS_ENABLE_SATELLITE "Build the independent satellite Doppler core" ON)
include(CTest)

if(BRICOHAMS_ENABLE_SATELLITE)
    add_library(bricohams_satellite STATIC bricohams/doppler.c bricohams/cat_rx.c bricohams/cat_tx.c)
    target_include_directories(bricohams_satellite PUBLIC bricohams)
    target_compile_features(bricohams_satellite PUBLIC c_std_99)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(bricohams_satellite PRIVATE -Wall -Wextra -Wconversion -Werror)
    endif()
    if(BUILD_TESTING)
        add_executable(bricohams_cat_rx_test tests/cat_rx_test.c)
        target_link_libraries(bricohams_cat_rx_test PRIVATE bricohams_satellite)
        add_test(NAME satellite_cat_rx COMMAND bricohams_cat_rx_test)
        add_executable(bricohams_cat_tx_test tests/cat_tx_test.c)
        target_link_libraries(bricohams_cat_tx_test PRIVATE bricohams_satellite)
        add_test(NAME satellite_cat_tx COMMAND bricohams_cat_tx_test)
        add_executable(bricohams_doppler_test tests/doppler_test.c)
        target_link_libraries(bricohams_doppler_test PRIVATE bricohams_satellite)
        add_test(NAME satellite_doppler COMMAND bricohams_doppler_test)
        add_executable(bricohams_satellite_adapter_test tests/satellite_adapter_test.c)
        target_include_directories(bricohams_satellite_adapter_test PRIVATE . tests/stubs)
        target_link_libraries(bricohams_satellite_adapter_test PRIVATE bricohams_satellite)
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(bricohams_cat_rx_test PRIVATE -Wall -Wextra -Wconversion -Werror)
            target_compile_options(bricohams_cat_tx_test PRIVATE -Wall -Wextra -Wconversion -Werror)
            target_compile_options(bricohams_satellite_adapter_test PRIVATE -Wall -Wextra -Wconversion -Werror)
        endif()
        add_test(NAME satellite_adapter COMMAND bricohams_satellite_adapter_test)
    endif()
endif()

if(BRICOHAMS_ENABLE_ARDF)
    add_library(bricohams_ardf STATIC bricohams/ardf.c)
    target_include_directories(bricohams_ardf PUBLIC bricohams)
    target_compile_features(bricohams_ardf PUBLIC c_std_99)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(bricohams_ardf PRIVATE -Wall -Wextra -Wconversion -Werror)
    endif()

    if(BUILD_TESTING)
        add_executable(bricohams_ardf_test tests/ardf_test.c)
        target_link_libraries(bricohams_ardf_test PRIVATE bricohams_ardf)
        add_test(NAME ardf_core COMMAND bricohams_ardf_test)
        add_executable(bricohams_ardf_adapter_test tests/ardf_adapter_test.c)
        target_include_directories(bricohams_ardf_adapter_test PRIVATE . tests/stubs)
        target_link_libraries(bricohams_ardf_adapter_test PRIVATE bricohams_ardf)
        add_test(NAME ardf_adapter COMMAND bricohams_ardf_adapter_test)
    endif()

endif()

if(BRICOHAMS_ENABLE_SPECTRUM)
    add_library(bricohams_spectrum STATIC bricohams/spectrum.c bricohams/presets.c)
    target_include_directories(bricohams_spectrum PUBLIC bricohams)
    target_compile_features(bricohams_spectrum PUBLIC c_std_99)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(bricohams_spectrum PRIVATE -Wall -Wextra -Wconversion -Werror)
    endif()
    if(BUILD_TESTING)
        add_executable(bricohams_spectrum_test tests/spectrum_test.c)
        target_link_libraries(bricohams_spectrum_test PRIVATE bricohams_spectrum)
        add_test(NAME spectrum_core COMMAND bricohams_spectrum_test)
        add_executable(bricohams_presets_test tests/presets_test.c)
        target_link_libraries(bricohams_presets_test PRIVATE bricohams_spectrum)
        add_test(NAME spectrum_presets COMMAND bricohams_presets_test)
        add_executable(bricohams_spectrum_adapter_test tests/spectrum_adapter_test.c)
        target_include_directories(bricohams_spectrum_adapter_test PRIVATE . tests/stubs)
        target_link_libraries(bricohams_spectrum_adapter_test PRIVATE bricohams_spectrum)
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(bricohams_spectrum_adapter_test PRIVATE -Wall -Wextra -Wconversion -Werror)
        endif()
        add_test(NAME spectrum_adapter COMMAND bricohams_spectrum_adapter_test)
    endif()
endif()
