cmake_minimum_required(VERSION 3.22)
project(bricohams_platform LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_library(bh_package STATIC bh_package.c)
add_library(bh_launcher STATIC bh_launcher.c)
add_library(bh_store STATIC bh_store.c)
add_library(bh_transfer STATIC bh_transfer.c)
add_library(bh_stream STATIC bh_stream.c)
target_link_libraries(bh_stream PUBLIC bh_transfer)
target_link_libraries(bh_transfer PUBLIC bh_package)
target_link_libraries(bh_store PUBLIC bh_package)
target_include_directories(bh_launcher PUBLIC .)
target_include_directories(bh_package PUBLIC .)
target_compile_features(bh_package PUBLIC c_std_99)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wconversion -Werror)
    target_compile_options(bh_package PRIVATE -Wall -Wextra -Wconversion -Werror)
    target_compile_options(bh_launcher PRIVATE -Wall -Wextra -Wconversion -Werror)
    target_compile_options(bh_store PRIVATE -Wall -Wextra -Wconversion -Werror)
    target_compile_options(bh_transfer PRIVATE -Wall -Wextra -Wconversion -Werror)
    target_compile_options(bh_stream PRIVATE -Wall -Wextra -Wconversion -Werror)
endif()
if(BUILD_TESTING)
    add_executable(bh_package_test package_test.c)
    target_link_libraries(bh_package_test PRIVATE bh_package)
    add_test(NAME bh_package COMMAND bh_package_test)
    add_executable(bh_diagnostics_test diagnostics.c diagnostics_test.c)
    target_include_directories(bh_diagnostics_test PRIVATE .)
    add_test(NAME bh_diagnostics COMMAND bh_diagnostics_test)
    add_executable(bh_blocks_test blocks.c blocks_test.c)
    target_include_directories(bh_blocks_test PRIVATE .)
    add_test(NAME bh_blocks COMMAND bh_blocks_test)
    add_executable(bh_launcher_test launcher_test.c diagnostics.c blocks.c)
    target_link_libraries(bh_launcher_test PRIVATE bh_launcher)
    add_test(NAME bh_launcher COMMAND bh_launcher_test)
    add_executable(bh_store_test store_test.c)
    target_link_libraries(bh_store_test PRIVATE bh_store)
    add_test(NAME bh_store COMMAND bh_store_test)
    add_executable(bh_shell_test shell_test.c bh_shell.c bh_text.c diagnostics.c blocks.c)
    target_link_libraries(bh_shell_test PRIVATE bh_launcher)
    add_test(NAME bh_shell COMMAND bh_shell_test)
    add_executable(bh_transfer_test transfer_test.c)
    target_link_libraries(bh_transfer_test PRIVATE bh_transfer)
    add_test(NAME bh_transfer COMMAND bh_transfer_test)
    add_executable(bh_stream_test stream_test.c)
    target_link_libraries(bh_stream_test PRIVATE bh_stream)
    add_test(NAME bh_stream COMMAND bh_stream_test)
    find_program(BH_NODE_EXECUTABLE node)
    if(BH_NODE_EXECUTABLE)
        add_executable(bh_transfer_bridge transfer_bridge.c)
        target_link_libraries(bh_transfer_bridge PRIVATE bh_stream bh_store)
        add_test(NAME bh_transfer_integration COMMAND "${BH_NODE_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/transfer_integration.cjs" "$<TARGET_FILE:bh_transfer_bridge>")
    endif()
endif()
